**Best one-many relationship b/w Stock(parent) & Stockhistory(Child)**

class Stock(db.Model):
    __tablename__ = 'stock'
    #Define the one-to-many relationship with StockHistory
    s_history = db.relationship('StockHistory', back_populates='item')

class StockHistory(db.Model):
    __tablename__ = 'stock_history'
    id = db.Column(db.Integer, primary_key=True, autoincrement=True)
    user_id = db.Column(db.Integer, default='0', nullable=True)

    # Define a foreign key to link to the parent Item
    item_id = db.Column(db.Integer, db.ForeignKey('stock.id'), nullable=False)
    #Define the relationship back to Item
    item = db.relationship('Item', back_populates='s_history')
